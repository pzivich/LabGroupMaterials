\begin{frame}{Public-Access}
	You or your co-authors may receive NIH, NSF, or other funding
	\begin{itemize}
		\item That funding may be from tax-payers
		\item So obligation to make our work available to the public
		\begin{itemize}
			\item Done through PMC (PubMed Central)
		\end{itemize}
		\item Important for grant progress reports
		\begin{itemize}
			\item Only those in the PMC process `count'
		\end{itemize}
	\end{itemize}
	~\\
	Done though NIHMS (NIH Manuscript Submission)
\end{frame}

\begin{frame}{Initiating NIHMS Process}

		\item If publishing open-access some journals will
	\end{itemize}
	~\\
	If journals do \textit{not} start the process, can initiate from PubMed
